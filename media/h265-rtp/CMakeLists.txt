cmake_minimum_required(VERSION 3.20)
project(emberbsd_h265_rtp VERSION 1.0 LANGUAGES C)
include(GNUInstallDirs)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-app-1.0 gstreamer-video-1.0)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(E35_WARNINGS_AS_ERRORS "Reject warnings in the example consumers" ON)
foreach(program IN ITEMS rtp-fault-relay check-frames ffmpeg-h265-receive gst-h265-rtp)
  add_executable(${program} ${program}.c)
  target_compile_options(${program} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(E35_WARNINGS_AS_ERRORS)
    target_compile_options(${program} PRIVATE -Werror)
  endif()
  set_target_properties(${program} PROPERTIES
    INSTALL_RPATH "${CMAKE_INSTALL_PREFIX}/lib;${CMAKE_INSTALL_PREFIX}/lib/ffmpeg9;${CMAKE_INSTALL_PREFIX}/gcc16/lib")
  install(TARGETS ${program} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endforeach()
target_link_libraries(ffmpeg-h265-receive PRIVATE PkgConfig::FFMPEG)
target_link_libraries(gst-h265-rtp PRIVATE PkgConfig::GST)
install(PROGRAMS make-fixture.rb check-rtp.rb test-checker.rb
  DESTINATION ${CMAKE_INSTALL_DATADIR}/emberbsd-h265-rtp)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/ember-h265-rtp)
