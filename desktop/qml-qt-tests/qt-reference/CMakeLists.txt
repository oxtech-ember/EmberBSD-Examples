cmake_minimum_required(VERSION 3.23)
project(qml_qt_reference LANGUAGES CXX)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Quick Test)
add_executable(qt_reference reference.cpp)
target_link_libraries(qt_reference PRIVATE Qt6::Quick)
target_compile_features(qt_reference PRIVATE cxx_std_17)
add_executable(qt_mousearea mousearea.cpp)
target_link_libraries(qt_mousearea PRIVATE Qt6::Quick Qt6::Test)
target_compile_features(qt_mousearea PRIVATE cxx_std_17)
add_executable(qt_behavior behavior.cpp)
target_link_libraries(qt_behavior PRIVATE Qt6::Quick Qt6::Test)
target_compile_features(qt_behavior PRIVATE cxx_std_17)
add_executable(qt_visual_opacity visual_opacity.cpp)
target_link_libraries(qt_visual_opacity PRIVATE Qt6::Quick Qt6::Test)
target_compile_features(qt_visual_opacity PRIVATE cxx_std_17)
if(APPLE)
    target_compile_options(qt_visual_opacity PRIVATE -stdlib=libc++)
    target_link_options(qt_visual_opacity PRIVATE -stdlib=libc++)
    target_compile_options(qt_behavior PRIVATE -stdlib=libc++)
    target_link_options(qt_behavior PRIVATE -stdlib=libc++)
    target_compile_options(qt_mousearea PRIVATE -stdlib=libc++)
    target_link_options(qt_mousearea PRIVATE -stdlib=libc++)
    target_compile_options(qt_reference PRIVATE -stdlib=libc++)
    target_link_options(qt_reference PRIVATE -stdlib=libc++)
endif()
